on_actions = {
	# Private Military Company spirit: +300 manpower every month
	on_monthly_WAG = {
		effect = {
			# African Contracts: recount the countries with Russian influence above 50
			if = {
				limit = { has_dynamic_modifier = { modifier = WAG_african_contracts } }
				WAG_update_african_contracts = yes
			}
			if = {
				limit = { has_idea = WAG_private_military_company }
				add_manpower = 300
			}
		}
	}

	# Operation Aibolit coalition: dismantled once the war against the CPC is over
	on_weekly_WAG = {
		effect = {
			# The Storm in Europe completes itself when the First European War begins
			if = {
				limit = {
					has_focus_tree = WAG_bangui_focus_tree
					has_global_flag = SOV_nato_war_begins_global
					NOT = { has_completed_focus = WAG_bng_storm }
				}
				complete_national_focus = WAG_bng_storm
			}
			# Stabilise the Country completes itself once the CAR civil war is over
			if = {
				limit = {
					has_focus_tree = WAG_car_focus_tree
					NOT = { has_completed_focus = WAG_car_stabilise_country }
					WAG_car_civil_war_over = yes
				}
				complete_national_focus = WAG_car_stabilise_country
			}
			if = {
				limit = {
					has_country_flag = WAG_car_coalition
					NOT = { has_war_with = CPC }
				}
				clr_country_flag = WAG_car_coalition
				if = {
					limit = { is_faction_leader = yes }
					dismantle_faction = yes
				}
			}
		}
	}
}
