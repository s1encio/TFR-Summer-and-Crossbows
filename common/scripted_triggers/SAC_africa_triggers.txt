# Countries that can open the Africa panel (only after Wagner completes 'Stabilise the Country')
SAC_can_see_africa_panel = {
	has_global_flag = SAC_africa_unlocked
	OR = {
		tag = WAG
		AND = { tag = SOV NOT = { has_global_flag = WAG_mutiny_failed } }
		tag = PRC
		tag = FRA
		tag = USA
	}
}

# Which influence slot the country belongs to (Wagner counts as Russia until a failed mutiny)
SAC_afr_slot_SOV = { OR = { tag = WAG AND = { tag = SOV NOT = { has_global_flag = WAG_mutiny_failed } } } }
SAC_afr_slot_PRC = { tag = PRC }
SAC_afr_slot_EUR = { tag = FRA }
SAC_afr_slot_USA = { tag = USA }
# Own influence in a country is at least 30 / 60 / 80
SAC_afr_inf_AO_30 = { OR = { AND = { SAC_afr_slot_SOV = yes check_variable = { global.AFR_AO_SOV > 29 } } AND = { SAC_afr_slot_PRC = yes check_variable = { global.AFR_AO_PRC > 29 } } AND = { SAC_afr_slot_EUR = yes check_variable = { global.AFR_AO_EUR > 29 } } AND = { SAC_afr_slot_USA = yes check_variable = { global.AFR_AO_USA > 29 } } } }
SAC_afr_inf_AO_60 = { OR = { AND = { SAC_afr_slot_SOV = yes check_variable = { global.AFR_AO_SOV > 59 } } AND = { SAC_afr_slot_PRC = yes check_variable = { global.AFR_AO_PRC > 59 } } AND = { SAC_afr_slot_EUR = yes check_variable = { global.AFR_AO_EUR > 59 } } AND = { SAC_afr_slot_USA = yes check_variable = { global.AFR_AO_USA > 59 } } } }
