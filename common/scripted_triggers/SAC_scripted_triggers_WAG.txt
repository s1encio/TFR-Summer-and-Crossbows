# Must be an EU or NATO member and not aligned with Russia.

	exists = yes
	NOT = { tag = WAG }

	NOT = { is_subject_of = SOV }
	NOT = { is_in_faction_with = SOV }

		is_EU_member = yes

		is_in_faction_with = USA
		is_in_faction_with = ENG
		AND = {
			tag = USA
			is_faction_leader = yes
		}
		AND = {
			tag = ENG
			is_faction_leader = yes
		}
	}
}
