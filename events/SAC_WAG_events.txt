# Sent to Sudan by Ask for Participation of Sudan (FROM = Wagner)
country_event = {
	id = sac_wag.10
	title = sac_wag.10.t
	desc = sac_wag.10.d
	picture = GFX_russians_army_streets

	is_triggered_only = yes

	# Send troops to the CAR government
	option = {
		name = sac_wag.10.a
		ai_chance = { factor = 70 }
		if = {
			limit = { country_exists = CAR }
			CAR = {
				division_template = {
					name = "Sudanese Expeditionary Brigade"
					regiments = {
						motorized = { x = 0 y = 0 }
						motorized = { x = 0 y = 1 }
						infantry = { x = 1 y = 0 }
					}
					support = {
						engineer = { x = 0 y = 0 }
					}
				}
				capital_scope = {
					create_unit = {
						division = "name = \"Sudanese Expeditionary Brigade 1\" division_template = \"Sudanese Expeditionary Brigade\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
						owner = CAR
					}
					create_unit = {
						division = "name = \"Sudanese Expeditionary Brigade 2\" division_template = \"Sudanese Expeditionary Brigade\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
						owner = CAR
					}
				}
			}
		}
		FROM = { country_event = { id = sac_wag.11 days = 1 } }
	}

	# Refuse
	option = {
		name = sac_wag.10.b
		ai_chance = { factor = 30 }
		FROM = { country_event = { id = sac_wag.12 days = 1 } }
	}
}

# Mali coup (TFR event mali.1 sets MLI_Civil_war_flag): the rebel coalition in the CAR
# receives three new divisions. Hidden, fires once, a day or so after the coup.
country_event = {
	id = sac_wag.20
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = CPC
		country_exists = MLI
		MLI = { has_country_flag = MLI_Civil_war_flag }
	}

	mean_time_to_happen = { days = 1 }

	immediate = {
		division_template = {
			name = "Coalition Assault Group"
			regiments = {
				special_forces = { x = 0 y = 0 }
				special_forces = { x = 0 y = 1 }
				infantry = { x = 1 y = 0 }
				infantry = { x = 1 y = 1 }
			}
			support = {
				anti_tank = { x = 0 y = 0 }
				anti_air = { x = 0 y = 1 }
				artillery = { x = 0 y = 2 }
			}
		}
		if = {
			limit = { capital_scope = { is_controlled_by = ROOT } }
			capital_scope = {
				create_unit = {
					division = "name = \"Coalition Assault Group 1\" division_template = \"Coalition Assault Group\" start_experience_factor = 0.5 start_equipment_factor = 1.0"

					division = "name = \"Coalition Assault Group 2\" division_template = \"Coalition Assault Group\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
					allow_spawning_on_enemy_provs = no
					owner = ROOT
				}
				create_unit = {
					division = "name = \"Coalition Assault Group 3\" division_template = \"Coalition Assault Group\" start_experience_factor = 0.5 start_equipment_factor = 1.0"

			random_owned_controlled_state = {

					owner = ROOT
				}
				create_unit = {
					division = "name = \"Coalition Assault Group 2\" division_template = \"Coalition Assault Group\" start_experience_factor = 0.5 start_equipment_factor = 1.0"
					allow_spawning_on_enemy_provs = no
