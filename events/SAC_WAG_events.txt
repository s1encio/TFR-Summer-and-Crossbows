				limit = { is_faction_leader = yes }

			}
			if = {
				limit = { is_in_faction = yes }
				leave_faction = yes
			}

			custom_effect_tooltip = sac_wag.31.csto_tt
		}
		country_event = { id = sac_wag.32 days = 2 }
	}
}

			tree = WAG_bangui_focus_tree

		name = sac_wag.32.b
