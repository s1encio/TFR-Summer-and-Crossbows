# Sent to Sudan by Ask for Participation of Sudan (FROM = Wagner)
country_event = {
	id = sac_wag.10
	title = sac_wag.10.t
	desc = sac_wag.10.d
	picture = GFX_russians_army_streets

	is_triggered_only = yes

	# Send troops to the CAR government
	option = {
		name = sac_wag.10.a
		ai_chance = { factor = 70 }
		# Real Sudanese troops: owned by Sudan, spawned in Bangui (state 778), with military
		# access from the CAR, and Sudan joins the fight against the CPC.
		# (create_unit only works in a STATE scope; HOI4 has no script effect for expeditionary forces.)
		if = {
			limit = { country_exists = CAR }
			division_template = {
				name = "Sudanese Expeditionary Brigade"
				regiments = {
					motorized = { x = 0 y = 0 }
					motorized = { x = 0 y = 1 }
					infantry = { x = 1 y = 0 }
				}
				support = {
					engineer = { x = 0 y = 0 }
				}
			}
			CAR = { give_military_access = SUD }
			778 = {
				create_unit = {
					division = "name = \"Sudanese Expeditionary Brigade 1\" division_template = \"Sudanese Expeditionary Brigade\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
					owner = SUD
					allow_spawning_on_enemy_provs = no
				}
				create_unit = {
					division = "name = \"Sudanese Expeditionary Brigade 2\" division_template = \"Sudanese Expeditionary Brigade\" start_experience_factor = 0.4 start_equipment_factor = 1.0"
					owner = SUD
					allow_spawning_on_enemy_provs = no
				}
			}
			if = {
				limit = {
					country_exists = CPC
					NOT = { has_war_with = CPC }
				}
				declare_war_on = { target = CPC type = annex_everything }
			}
			custom_effect_tooltip = sac_wag.10.a_tt
		}
		FROM = { country_event = { id = sac_wag.11 days = 1 } }
	}

	# Refuse
	option = {
		name = sac_wag.10.b
		ai_chance = { factor = 30 }
		FROM = { country_event = { id = sac_wag.12 days = 1 } }
	}
}

# Mali coup = TFR event mali.9 ("2020 Coup", fired ~day 232 from TFR on_startup).
# It sets no flag, so we detect it by its result: Goita in power / nationalist military government.
# (mali.1 is NOT the coup: it is the war with the Islamic State in the Sahel.)
# Step 1: Wagner notices the coup and schedules step 2 one to two weeks later.
country_event = {
	id = sac_wag.20
	hidden = yes
	fire_only_once = yes

	trigger = {
		tag = WAG
		country_exists = MLI
		MLI = {
			OR = {
				has_country_leader = { character = MLI_Assimi_Goita_char ruling_only = yes }
				has_government = nationalist
			}
		}
		NOT = { has_completed_focus = WAG_call_from_moscow }
	}

	is_triggered_only = yes  # fired every week by on_weekly_WAG (SAC_on_actions_WAG.txt)

	immediate = {
		country_event = { id = sac_wag.21 days = 7 random_days = 7 }
	}
}
