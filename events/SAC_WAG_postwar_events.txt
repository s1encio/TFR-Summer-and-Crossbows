# ---------- The March of Justice (focus WAG_btr_march) ----------
country_event = {
	id = sac_wpw.60
	title = sac_wpw.60.t
	desc = sac_wpw.60.d
	picture = GFX_SOV_wagner_putsch_1
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_country_flag = WAG_mutiny_done
			set_temp_variable_to_random = { var = WAG_mutiny_roll min = 0 max = 100 integer = yes }
			if = {
				limit = { check_variable = { WAG_mutiny_roll < WAG_mutiny_ready } }
				set_country_flag = WAG_mutiny_will_win
			}
		}
	}

	option = {
		name = sac_wpw.60.a
		custom_effect_tooltip = WAG_mutiny_chance_tt
		hidden_effect = {
			if = {
				limit = { has_country_flag = WAG_mutiny_will_win }
				country_event = { id = sac_wpw.61 days = 3 }
			}
			else = {
				country_event = { id = sac_wpw.62 days = 3 }
			}
		}
	}
}

# Russia (now played by the former Wagner player) absorbs Wagner; the CAR becomes a puppet
country_event = {
	id = sac_wpw.63
	title = sac_wpw.63.t
	desc = sac_wpw.63.d
	picture = GFX_SOV_wagner_flags
	is_triggered_only = yes

	option = {
		name = sac_wpw.63.a
		if = {
			limit = { country_exists = WAG }
			annex_country = { target = WAG transfer_troops = yes }
		}
		if = {
			limit = { NOT = { country_exists = CAR } }
			release_puppet = CAR
		}
	}
}
